Пример функции Информация()

Функция Информация() Экспорт     

СтруктураОписания 		= CW_ПодключаемыеКомандыСервер.СведенияОПодключаемомОбъекте("3.2.0.1");
СтруктураОписания.Тип 	= CW_ПодключаемыеКомандыСервер.Тип<>();                        //Тип внешнего объекта
СтруктураОписания.Вид 	= CW_ПодключаемыеКомандыСервер.ВидОбработки<>();      //Вид внешней обработки/отчета
СтруктураОписания.Наименование				= "<Наименование подключаемого объекта>";
СтруктураОписания.Описание					= "<Описание подключаемого объекта>";
СтруктураОписания.Версия					        = "3.2.Х.Х ";       //Версия внешнего объекта;
СтруктураОписания.МинимальнаяВерсияМодуля	= "8.3.10.10";     //Минимальная версия модуля БЭ, с которой гарантирована работа обработки, не ниже 8.3.10.10!
	
СтруктураОписания.РасполагатьНаРабочемСтоле	= Истина;  //Признак расположения команды на рабочем столе модуля;
СтруктураОписания.ИсторияИзменений			= Истина;  //Признак наличия истории изменений обработки;
СтруктураОписания.Настройка					= Истина;  //Признак наличия дополнительных настроек обработки.	
	//Методы
	CW_ПодключаемыеКомандыСервер.УстановитьМетодыВнешнегоОбъекта(СтруктураОписания, <ИмяМетода>);
	//Методы	
	//Формы
	CW_ПодключаемыеКомандыСервер.УстановитьИмяФормыОбъекта(СтруктураОписания, <ИмяФормыОбъекта>);   
	CW_ПодключаемыеКомандыСервер.УстановитьИмяФормыНастроекОбъекта(СтруктураОписания, <ИмяФормыНастроек>);   //В случае если СтруктураОписания.Настройка= Истина
	//Формы	
	//Команды
	CW_ПодключаемыеКомандыСервер.УстановитьПараметрыВнешнейКоманды(СтруктураОписания, 
										<ИмяОбъектаМетаданных>, 
										<ИмяФормы>,
										,
										,
										,
										<Представление команды>);
										
	CW_ПодключаемыеКомандыСервер.УстановитьКомандыВнешнегоОбъекта(СтруктураОписания,
					<ИмяКоманды>,
					<ПредставлениеКоманды>,
					БиблиотекаКартинок.<ИмяКартинки>,
					Истина);
	//Команды

Возврат СтруктураОписания;

КонецФункции
