<!-- Начало вставки - скрипт обработки прокрутки -->
<a id="1C_bridge" href="#" style="position:absolute; width:1px; height:1px; opacity:0;">.</a>

<script>
(function() {
    function initScrollTracking() {
        var scrollTimeout;
        const topBoundary = window.innerHeight * 0.7;
        // Удаляем прошлый обработчик, если он существовал
        if (window._1cScrollHandler) {
            window.removeEventListener('scroll', window._1cScrollHandler, true);
        }
        // Добавляем обработчик
        window._1cScrollHandler = function() {
            clearTimeout(scrollTimeout);
            scrollTimeout = setTimeout(findCurrentId, 200);
        };
        window.addEventListener('scroll', window._1cScrollHandler, true);
        
        // Сбрасываем кэш Id
        window._1cLastSentId = "";

        function findCurrentId() {
            const anchors = document.querySelectorAll('[id]');
            var currentId = "";
            for (const anchor of anchors) {
                const id = anchor.id;
                if (!id || id.startsWith('1C_')) {
                    continue;
                }
                const rect = anchor.getBoundingClientRect();
                if (rect.top <= topBoundary) {
                    currentId = id;
                    // Идентификатор в видимой области или сразу над ней - он нам и нужен 
                    if (rect.top >= -10) {
                        break;
                    }
                } else {
                    // Идентификатор ниже topBoundary - возвращаем предыдущий 
                    break;
                }
            }
            if (currentId) {
                sendClickTo1C(currentId);
            }
        }

        function sendClickTo1C(id) {
            if (!id || id === window._1cLastSentId) return;
            const bridge = document.getElementById('1C_bridge');
            if (bridge) {
                window._1cLastSentId = id;
                bridge.href = "Scroll=" + id;
                bridge.click();
            }
        }
    }

    // Запускаем отслеживание для текущей строки
    initScrollTracking();

    // Настраиваем отслеживание обновления DOM.
    // Как только содержание HTML заменится на новое, скрипт перезапустится.
    if (!window._1cMutationObserver) {
        window._1cMutationObserver = new MutationObserver(function() {
            // Структура документа изменилась — заново инициализируем скролл
            initScrollTracking();
        });
        
        window._1cMutationObserver.observe(document.documentElement, { 
            childList: true, 
            subtree: true 
        });
    }
})();
</script>
<!-- Конец вставки - скрипт обработки прокрутки -->
